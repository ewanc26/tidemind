cmake_minimum_required(VERSION 3.20)
project(Tidemind VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(TIDEMIND_GUI "Build the SDL2 desktop game" ON)
option(TIDEMIND_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
add_library(tidemind_core src/simulation.cpp src/neural.cpp)
target_include_directories(tidemind_core PUBLIC src)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(tidemind_core PRIVATE -Wall -Wextra -Wpedantic)
  if(TIDEMIND_SANITIZE)
    target_compile_options(tidemind_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(tidemind_core PUBLIC -fsanitize=address,undefined)
  endif()
endif()
if(TIDEMIND_GUI)
  find_package(SDL2 2.0.18 REQUIRED CONFIG)
  find_package(SDL2_ttf REQUIRED CONFIG)
  add_executable(tidemind src/main.cpp src/ui.cpp)
  target_link_libraries(tidemind PRIVATE tidemind_core SDL2::SDL2 SDL2_ttf::SDL2_ttf)
  target_compile_definitions(tidemind PRIVATE TIDEMIND_ASSETS="${CMAKE_CURRENT_SOURCE_DIR}/assets")
  if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(tidemind PRIVATE -Wall -Wextra -Wpedantic)
  endif()
  add_custom_command(TARGET tidemind POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory
    "${CMAKE_CURRENT_SOURCE_DIR}/assets" "$<TARGET_FILE_DIR:tidemind>/assets")
  install(TARGETS tidemind RUNTIME DESTINATION bin)
  install(DIRECTORY assets DESTINATION share/tidemind)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(tidemind_tests tests/core_tests.cpp)
  target_link_libraries(tidemind_tests PRIVATE tidemind_core)
  add_test(NAME simulation_and_neural COMMAND tidemind_tests)
  if(TIDEMIND_GUI)
    add_test(NAME interface_flow COMMAND tidemind --smoke-test --save ${CMAKE_CURRENT_BINARY_DIR}/smoke.save)
    set_tests_properties(interface_flow PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy" TIMEOUT 30)
    add_test(NAME command_help COMMAND tidemind --help)
  endif()
endif()
